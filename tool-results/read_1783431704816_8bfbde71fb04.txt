    20→const persianDigits = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"];
    21→
    22→function toPersian(n: number): string {
    23→  return n.toString().replace(/\d/g, (d) => persianDigits[+d]);
    24→}
    25→
    26→function formatPersian(n: number): string {
    27→  if (n === 0) return "۰";
    28→  return toPersian(n).replace(/\B(?=(\d{3})+(?!\d))/g, "٬");
    29→}

    35→interface ColumnState {

    37→  lower: [boolean, boolean, boolean, boolean];

    41→

    44→  b: number;
    45→  op: "+" | "-";
    46→  answer: number;

    49→/* ═══════════════════════════════════════════════════

    52→
    53→const NUM_COLS = 13;
    54→
    55→function createInitialColumns(): ColumnState[] {
    56→  return Array.from({ length: NUM_COLS }, () => ({
    57→    upper: false,
    58→    lower: [false, false, false, false],
    59→  }));
    60→}
    61→
    62→function getColumnValue(col: ColumnState): number {
    63→  return (col.upper ? 5 : 0) + col.lower.filter(Boolean).length;
    64→}
    65→
    66→function getAbacusTotal(columns: ColumnState[]): number {
    67→  return columns.reduce(
    68→    (sum, col, i) => sum + getColumnValue(col) * 10 ** i,
    69→    0
    70→  );
    71→}
    72→
    73→function generateProblem(): Problem {
    74→  const op: "+" | "-" = Math.random() > 0.5 ? "+" : "-";
    75→  let a = Math.floor(Math.random() * 900) + 100;
    76→  let b = Math.floor(Math.random() * 900) + 100;
    77→  if (op === "-" && a < b) [a, b] = [b, a];
    78→  return { a, b, op, answer: op === "+" ? a + b : a - b };
    79→}

   194→
   195→function GlassButton({
   196→  children,
   197→  onClick,
   198→  variant = "default",
   199→  icon: Icon,
   200→  className = "",
   201→}: {
   202→  children: React.ReactNode;
   203→  onClick?: () => void;
   204→  variant?: "default" | "emerald" | "amber" | "red";
   205→  icon?: React.ElementType;
   206→  className?: string;
   207→}) {
   208→  const base =
   209→    "inline-flex items-center justify-center gap-2 rounded-xl text-sm font-medium border transition-all duration-200 cursor-pointer select-none";
   210→  const variants: Record<string, string> = {
   211→    default:
   212→      "bg-gray-50 border-gray-200 text-slate-900/60 hover:bg-gray-100 hover:text-slate-900/80",
   213→    emerald:
   214→      "bg-emerald-500/15 border-emerald-500/25 text-emerald-300 hover:bg-emerald-500/25",
   215→    amber:
   216→      "bg-amber-500/15 border-amber-500/25 text-amber-300 hover:bg-amber-500/25",
   217→    red: "bg-red-500/15 border-red-500/25 text-red-300 hover:bg-red-500/25",
   218→  };
   219→  return (
   220→    <button
   221→      onClick={onClick}
   222→      className={`${base} ${variants[variant]} ${className}`}
   223→    >
   224→      {Icon && <Icon className="w-4 h-4" />}
   225→      {children}
   226→    </button>
   227→  );
   228→}
