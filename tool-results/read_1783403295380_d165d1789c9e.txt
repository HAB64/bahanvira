    20→    20→const persianDigits = ["۰", "۱", "۲", "۳", "۴", "۵", "۶", "۷", "۸", "۹"];
    21→    21→
    22→    22→function toPersian(n: number): string {
    23→    23→  return n.toString().replace(/\d/g, (d) => persianDigits[+d]);
    24→    24→}
    25→    25→
    26→    26→function formatPersian(n: number): string {
    27→    27→  if (n === 0) return "۰";
    28→    28→  return toPersian(n).replace(/\B(?=(\d{3})+(?!\d))/g, "٬");
    29→    29→}

    62→    62→function getColumnValue(col: ColumnState): number {
    63→    63→  return (col.upper ? 5 : 0) + col.lower.filter(Boolean).length;

    66→    66→function getAbacusTotal(columns: ColumnState[]): number {
    67→    67→  return columns.reduce(
    68→    68→    (sum, col, i) => sum + getColumnValue(col) * 10 ** i,
    69→    69→    0
    70→    70→  );
    71→    71→}

    73→    73→function generateProblem(): Problem {
    74→    74→  const op: "+" | "-" = Math.random() > 0.5 ? "+" : "-";
    75→    75→  let a = Math.floor(Math.random() * 900) + 100;
    76→    76→  let b = Math.floor(Math.random() * 900) + 100;
    77→    77→  if (op === "-" && a < b) [a, b] = [b, a];
    78→    78→  return { a, b, op, answer: op === "+" ? a + b : a - b };
    79→    79→}
